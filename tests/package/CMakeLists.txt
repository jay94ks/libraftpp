cmake_minimum_required(VERSION 3.20)
project(raftpp_package_consumer LANGUAGES CXX)
if(RAFTPP_EXPECT_PACKAGE_MISSING)
    find_package(raftpp CONFIG QUIET)
    if(raftpp_FOUND OR TARGET raftpp::raftpp)
        message(FATAL_ERROR "Expected missing dependencies to leave raftpp unavailable")
    endif()
    return()
endif()
find_package(raftpp CONFIG REQUIRED)
get_target_property(_imported raftpp::raftpp IMPORTED)
if(NOT _imported OR NOT TARGET raftpp::core OR TARGET raftpp::raft)
    message(FATAL_ERROR "Expected an installed C++ target without the C reference target")
endif()
# Installed modules expose only active public headers and their own targets.
get_target_property(_public_includes raftpp::core INTERFACE_INCLUDE_DIRECTORIES)
list(GET _public_includes 0 _public_include)
file(READ "${_public_include}/raftpp/config.hpp" _features)
if(_features MATCHES "#define RAFTPP_WITH_UV 1")
    if(NOT TARGET raftpp::uv OR NOT EXISTS "${_public_include}/raftpp/uv/uv_io.hpp")
        message(FATAL_ERROR "Enabled UV module is missing its target or public headers")
    endif()
else()
    if(TARGET raftpp::uv OR EXISTS "${_public_include}/raftpp/uv")
        message(FATAL_ERROR "Disabled UV module leaked into the installed package")
    endif()
endif()
if(_features MATCHES "#define RAFTPP_WITH_FIXTURE 0"
   AND EXISTS "${_public_include}/raftpp/fixture.hpp")
    message(FATAL_ERROR "Disabled fixture header leaked into the installed package")
endif()
foreach(private_header IN ITEMS byte.hpp election.hpp receive.hpp tracing.hpp disk.hpp file.hpp
        writer.hpp log_writer.hpp file_backend.hpp probe.hpp segment_buffer.hpp snapshot_format.hpp)
    if(EXISTS "${_public_include}/raftpp/detail/${private_header}")
        message(FATAL_ERROR "Private header was installed: ${private_header}")
    endif()
endforeach()

add_executable(raftpp_package_consumer main.cpp)
target_link_libraries(raftpp_package_consumer PRIVATE raftpp::raftpp)
set_target_properties(raftpp_package_consumer PROPERTIES CXX_EXTENSIONS OFF)
add_executable(raftpp_core_consumer core.cpp)
target_link_libraries(raftpp_core_consumer PRIVATE raftpp::core)
set_target_properties(raftpp_core_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(TARGET raftpp::uv)
    add_executable(raftpp_uv_consumer uv.cpp)
    target_link_libraries(raftpp_uv_consumer PRIVATE raftpp::uv)
    set_target_properties(raftpp_uv_consumer PROPERTIES CXX_EXTENSIONS OFF)
endif()

if(WIN32)
    foreach(module IN ITEMS core uv)
        if(TARGET raftpp::${module})
            get_target_property(_type raftpp::${module} TYPE)
            if(_type STREQUAL "SHARED_LIBRARY")
                add_custom_command(TARGET raftpp_package_consumer POST_BUILD
                    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                        "$<TARGET_FILE:raftpp::${module}>" "$<TARGET_FILE_DIR:raftpp_package_consumer>")
            endif()
        endif()
    endforeach()
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND raftpp_package_consumer)

add_test(NAME installed_core_consumer COMMAND raftpp_core_consumer)
if(TARGET raftpp::uv)
    add_test(NAME installed_uv_consumer COMMAND raftpp_uv_consumer)
endif()

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../modules" "${CMAKE_CURRENT_BINARY_DIR}/modules")
