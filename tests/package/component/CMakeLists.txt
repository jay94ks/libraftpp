cmake_minimum_required(VERSION 3.20)
project(raftpp_component_consumer LANGUAGES CXX)
find_package(raftpp CONFIG REQUIRED COMPONENTS ${RAFTPP_COMPONENT})
if(NOT TARGET raftpp::core)
    message(FATAL_ERROR "Mandatory core is missing")
endif()
if(RAFTPP_COMPONENT STREQUAL "core")
    add_executable(component_consumer ../core.cpp)
else()
    add_executable(component_consumer ../../modules/${RAFTPP_COMPONENT}.cpp)
endif()
target_link_libraries(component_consumer PRIVATE raftpp::${RAFTPP_COMPONENT})
set_target_properties(component_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME component_consumer COMMAND component_consumer)
