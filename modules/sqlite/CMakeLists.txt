include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/dependency-module.cmake")
raftpp_add_dependency_module(sqlite)
find_package(SQLite3 3.24 REQUIRED)
find_package(Threads REQUIRED)
target_link_libraries(raftpp_sqlite PRIVATE SQLite::SQLite3 Threads::Threads)
target_include_directories(raftpp_sqlite PRIVATE "${PROJECT_SOURCE_DIR}/modules/core/privates")
raftpp_dependency_package("find_dependency(SQLite3 3.24)")
raftpp_dependency_package("find_dependency(Threads)")
raftpp_finish_dependency_module(sqlite)
