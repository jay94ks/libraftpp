find_package(Threads REQUIRED)
# P/Invoke needs a shared library even when core/uv are static.
add_library(raftpp_bindings SHARED src/bindings.cpp src/runtime.cpp)
add_library(raftpp::bindings ALIAS raftpp_bindings)
raftpp_configure_module(raftpp_bindings bindings raftpp/bindings/export.hpp RAFTPP_BINDINGS_API)
target_link_libraries(raftpp_bindings PRIVATE raftpp::core Threads::Threads ${CMAKE_DL_LIBS})
if(TARGET raftpp_uv)
    target_link_libraries(raftpp_bindings PRIVATE raftpp::uv)
endif()
foreach(module IN ITEMS asio sqlite rocksdb lz4 zstd zlib)
    if(TARGET raftpp_${module})
        target_link_libraries(raftpp_bindings PRIVATE raftpp::${module})
    endif()
endforeach()
set_target_properties(raftpp_bindings PROPERTIES INSTALL_RPATH "$ORIGIN")
raftpp_stage_headers("${CMAKE_CURRENT_SOURCE_DIR}/include/raftpp/bindings/bindings.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/raftpp/bindings/runtime.hpp")

if(RAFTPP_BINDINGS_NODEJS)
    find_path(RAFTPP_NODE_INCLUDE_DIR node_api.h PATH_SUFFIXES node nodejs/src REQUIRED)
    add_library(raftpp_nodejs MODULE src/nodejs.cpp)
    target_compile_features(raftpp_nodejs PRIVATE cxx_std_20)
    target_include_directories(raftpp_nodejs PRIVATE "${RAFTPP_NODE_INCLUDE_DIR}")
    target_link_libraries(raftpp_nodejs PRIVATE raftpp::bindings)
    set_target_properties(raftpp_nodejs PROPERTIES PREFIX "" SUFFIX ".node"
        OUTPUT_NAME raftpp_nodejs CXX_EXTENSIONS OFF INSTALL_RPATH "$ORIGIN")
    if(RAFTPP_INSTALL)
        install(TARGETS raftpp_nodejs LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
    endif()
endif()
